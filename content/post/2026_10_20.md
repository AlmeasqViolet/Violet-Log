Agent协助的配置经验，方便一键手动配置。

<!--more-->

## 1、修改镜像
```bash
# Curl
sudo install -d -m 755 /usr/local/bin /etc/makepkg.conf.d /etc/makepkg.d

sudo tee /usr/local/bin/makepkg-curl-ghproxy >/dev/null <<'EOF'
#!/bin/bash
set -euo pipefail
GH_PROXY="${GH_PROXY:-https://gh-proxy.com}"
REAL_CURL="${REAL_CURL:-/usr/bin/curl}"
rewrite_url() {
    local url=$1
    case "$url" in
        https://github.com/*|http://github.com/*|\
        https://raw.githubusercontent.com/*|\
        https://gist.github.com/*|\
        https://gist.githubusercontent.com/*|\
        https://codeload.github.com/*|\
        https://objects.githubusercontent.com/*|\
        https://github-releases.githubusercontent.com/*)
            printf '%s/%s' "$GH_PROXY" "$url" ;;
        *) printf '%s' "$url" ;;
    esac
}
args=()
for arg in "$@"; do args+=("$(rewrite_url "$arg")"); done
exec "$REAL_CURL" "${args[@]}"
EOF

#DLAGENTS drop-in
sudo chmod 755 /usr/local/bin/makepkg-curl-ghproxy

sudo tee /etc/makepkg.conf.d/ghproxy.conf >/dev/null <<'EOF'
DLAGENTS=('file::/usr/bin/curl -qgC - -o %o %u'
          'ftp::/usr/bin/curl -qgfC - --ftp-pasv --retry 3 --retry-delay 3 -o %o %u'
          'http::/usr/local/bin/makepkg-curl-ghproxy -qgb "" -fLC - --retry 3 --retry-delay 3 -o %o %u'
          'https::/usr/local/bin/makepkg-curl-ghproxy -qgb "" -fLC - --retry 3 --retry-delay 3 -o %o %u'
          'rsync::/usr/bin/rsync --no-motd -z %u %o'
          'scp::/usr/bin/scp -C %u %o')
EOF

#gitconfig
sudo tee /etc/makepkg.d/gitconfig >/dev/null <<'EOF'
[url "https://gh-proxy.com/https://github.com/"]
	insteadOf = https://github.com/
	insteadOf = git://github.com/
[url "https://gh-proxy.com/https://gist.github.com/"]
	insteadOf = https://gist.github.com/
EOF
```

## 2、撤销改动

```bash
# 删除配置文件
sudo rm -f /etc/makepkg.d/gitconfig \
           /etc/makepkg.conf.d/ghproxy.conf \
           /usr/local/bin/makepkg-curl-ghproxy

# 删除配置文件目录
sudo rmdir --ignore-fail-on-non-empty /etc/makepkg.d /etc/makepkg.conf.d

# 验证是否删除
ls -l /usr/local/bin/makepkg-curl-ghproxy \
      /etc/makepkg.conf.d/ghproxy.conf \
      /etc/makepkg.d/gitconfig
```
